import bpy,numpy as np,math
from collections import Counter
from mathutils import Vector
from mathutils.bvhtree import BVHTree
def prepare_joints(body):
    old=body.data;old.update();n=len(old.vertices)
    verts=[tuple(v.co) for v in old.vertices];faces=[list(f.vertices) for f in old.polygons]
    mats=[f.material_index for f in old.polygons]
    uv=[tuple(x.uv) for x in old.uv_layers.active.data]
    tree=BVHTree.FromPolygons([Vector(v) for v in verts],faces,all_triangles=True)
    clean=next(f for f in old.polygons if abs(f.center.x)<.08 and -.45<f.center.z<-.35 and f.center.y<-.15)
    white=tuple(sum((old.uv_layers.active.data[i].uv for i in clean.loop_indices),Vector((0,0)))/len(clean.loop_indices))
    arms={};cap_ids=[];protrusion=0
    for side,sign in [('L',1),('R',-1)]:
        group=body.vertex_groups['Part.Arm.'+side].index
        arm={v.index for v in old.vertices if any(g.group==group for g in v.groups)}
        arms[side]=list(arm)
        triangles=[f for f in faces if all(i in arm for i in f)]
        counts=Counter(tuple(sorted((a,b))) for f in triangles for a,b in zip(f,f[1:]+f[:1]))
        edges=[(a,b) for f in triangles for a,b in zip(f,f[1:]+f[:1]) if counts[tuple(sorted((a,b)))]==1]
        following=dict(edges);start=edges[0][0];loop=[start];cursor=following[start]
        while cursor!=start:loop.append(cursor);cursor=following[cursor]
        assert len(loop)==len(edges)
        center=sum((Vector(verts[i]) for i in loop),Vector())/len(loop)
        def inside(point):
            nonlocal protrusion
            for _ in range(20):
                hit,normal,_,_=tree.find_nearest(point);signed=(point-hit).dot(normal)
                if signed<-.0002:break
                point=point.lerp(center,.20)
            hit,normal,_,_=tree.find_nearest(point);protrusion=max(protrusion,max(0,(point-hit).dot(normal)))
            return point
        for moving,direction in [(True,-sign)]:
            if moving:previous=loop
            else:
                previous=[]
                for i in reversed(loop):
                    previous.append(len(verts));cap_ids.append(len(verts));verts.append(verts[i])
            source=[Vector(verts[i]) for i in previous]
            for progress in [.30,.65,.90,1.0]:
                ring=[]
                for v in source:
                    p=v.lerp(center,progress)
                    if not moving:p+=Vector((-sign*.16*math.sin(progress*math.pi/2),0,0))
                    p=inside(p);index=len(verts);verts.append(tuple(p));ring.append(index);cap_ids.append(index)
                    if moving:arms[side].append(index)
                for j in range(len(ring)):
                    k=(j+1)%len(ring);faces.append([previous[k],previous[j],ring[j],ring[k]]);uv.extend([white]*4);mats.append(0)
                previous=ring
    mesh=bpy.data.meshes.new('Original exterior with internal shoulder caps');mesh.from_pydata(verts,[],faces);mesh.update()
    for mat in old.materials:mesh.materials.append(mat)
    layer=mesh.uv_layers.new(name='UVMap')
    for i,tex in enumerate(uv):layer.data[i].uv=tex
    for f,mat in zip(mesh.polygons,mats):f.material_index=mat;f.use_smooth=True
    body.shape_key_clear();body.data=mesh
    assert np.max(np.abs(np.array(verts[:n])-np.array([tuple(v.co) for v in old.vertices])))==0
    return np.array(verts),arms,{'originalSurfaceVerticesUnmoved':n,'internalCapVertices':len(cap_ids),'maximumRestCapProtrusion':protrusion}
